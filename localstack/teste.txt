# Criar bucket S3
aws --endpoint-url=http://localhost:4566 s3 mb s3://shopping-images

# Criar fila SQS
aws --endpoint-url=http://localhost:4566 sqs create-queue --queue-name tasks-queue

# Criar tópico SNS
aws --endpoint-url=http://localhost:4566 sns create-topic --name tasks-topic

# Criar tabela DynamoDB
aws --endpoint-url=http://localhost:4566 dynamodb create-table --table-name Tasks --attribute-definitions AttributeName=id,AttributeType=S --key-schema AttributeName=id,KeyType=HASH --billing-mode PAY_PER_REQUEST
